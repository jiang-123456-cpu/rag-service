import logging
import os
from datetime import datetime

def init_rag_logger():
    if not os.path.exists("logs"):
        os.makedirs("logs")

    log_path = os.path.join("logs", log_name)

        ]
    )

logger = init_rag_logger()

import yaml
import os
from src.utils.log_util import logger

def load_yaml_config(config_path: str = "config/config.yaml") -> dict:
    if not os.path.exists(config_path):
        logger.error(f"配置文件不存在：{config_path}")
        raise FileNotFoundError("缺少RAG核心配置yaml")

## 3.4 八股4：Pydantic参数校验模型
```python
from pydantic import BaseModel, Field, field_validator

class RAGQueryInput(BaseModel):
    query: str = Field(description="用户提问文本")
    top_k: int = Field(default=3, description="检索返回文档数")
    threshold: float = Field(default=0.6, description="相似度过滤阈值")
    temperature: float = Field(default=0.7)

    @field_validator("top_k")
    def valid_topk(cls, v):
        if not 1 <= v <= 10:
            raise ValueError("top_k 范围必须1~10")
        return v

    @field_validator("threshold")
    def valid_threshold(cls, v):
        if not 0 <= v <= 1:
            raise ValueError("相似度阈值范围0~1")
        return v
```

## 3.6 八股6：全局异常捕获装饰器 decorator_util.py
```python
import functools
from src.utils.log_util import logger

def rag_catch_exception(func):
    @functools.wraps(func)
    def wrapper(*args, **kwargs):
        try:
            return func(*args, **kwargs)
        except Exception as err:
            logger.error(f"函数 {func.__name__} 运行异常: {str(err)}", exc_info=True)
            # 根据业务分支返回兜底值
            if "retrieve" in func.__name__:
                return []
            elif "generate" in func.__name__:
                return "大模型生成失败，请重新提问"
            return None
    return wrapper
```

## 3.7 八股7：向量归一化、余弦相似度 vector_util.py
```python
import numpy as np

def normalize_vector(vec: np.ndarray) -> np.ndarray:
    norm = np.linalg.norm(vec)
    if norm == 0:
        return vec
    return vec / norm

def cosine_similarity(vec_a: np.ndarray, vec_b: np.ndarray) -> float:
    a_norm = normalize_vector(vec_a)
    b_norm = normalize_vector(vec_b)
    sim = float(np.dot(a_norm, b_norm))
    return round(sim, 4)
```

## 3.8 八股8：滑动窗口文本切片 chunker.py
```python
from src.utils.log_util import logger

def split_text_chunk(text: str, chunk_size: int = 512, overlap: int = 60) -> list[str]:
    chunks = []
    text_len = len(text)
    start = 0
    while start < text_len:
        end = min(start + chunk_size, text_len)
        chunk = text[start:end]
        chunks.append(chunk)
        start = end - overlap
    logger.info(f"文本分割完成，总片段数：{len(chunks)}")
    return chunks
```

# 六、实验问题与总结
## 6.1 实验常见问题
1. yaml读取乱码：必须指定encoding="utf-8"；
2. 日志重复打印：项目仅初始化一次logger全局对象；
3. lru_cache无法缓存numpy数组：缓存仅用于文本query，向量结果不存入缓存；
4. 文本切片重叠过小：易丢失上下文，推荐overlap设置50~80。

## 6.2 实验总结
本套10组Python八股文覆盖RAG项目全部底层通用能力，标准化代码可直接复用至毕业设计、本地Demo、企业轻量化RAG系统。
统一模板带来3点收益：
1. 工程规范统一，多人协作无代码风格割裂；
2. 提前拦截参数、文件、模型调用异常，提升系统稳定性；
3. 工具函数高度解耦，后续更换FAISS/Chroma/ Milvus向量库无需重构底层基础代码。
